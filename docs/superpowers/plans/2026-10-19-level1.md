# MathTime Level 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Level 1 screen exactly as described in `docs/superpowers/specs/2026-10-19-level1-design.md`. It is a single-digit addition game with a fixed inline answer box, an on-screen keypad, 3 lives per problem, and the Blueprint theme in IBM Plex Mono, aligned to the real ink of the digits within half a pixel.

**Architecture:** A single static page (`index.html`) loads plain JavaScript modules. There is no framework and no build step. Game rules (`game.js`, `level1.js`) and alignment math (`ink.js`) are pure and unit-tested in Node. The browser-only parts are:
- `scanner.js`: measures ink on a canvas;
- `layout.js`: sizes, measures and snaps the screen;
- `view.js`: renders and animates;
- `input.js`: keypad and keyboard.

`main.js` wires these together. Playwright browser tests check the flows and measure alignment from real screenshot pixels, which automates the checks the review agents did by hand.

**Tech Stack:** HTML, CSS, JavaScript (ES modules). Node 24's built-in `node:test` for unit tests. `@playwright/test` and `pngjs` for browser tests. A zero-dependency Node static server. Fonts are IBM Plex Mono and IBM Plex Sans from Google Fonts.

**Decision for review:** The spec left the build approach open. This plan uses **no framework**, because the screen is one page, the verified mockup is already plain JavaScript, and a framework adds nothing Level 1 needs. If you'd rather use one, say so before Task 1.

**Visual reference:** `docs/superpowers/specs/2026-10-19-level1-mockup.html` is the reviewed prototype. Open it with no `?theme=` to see the target. The code below is a modular port of its Blueprint path. The other themes and the pixel-grid logic are left out on purpose.

**Rule that applies throughout:** never show the player-designer a UI state that a separate review agent hasn't rendered and measured (spec §8). Task 7 is the gate.

```js
// Tiny static file server for development and the browser tests.
import { createServer } from 'node:http';
import { readFile } from 'node:fs/promises';
import { extname, join, normalize } from 'node:path';
import { fileURLToPath } from 'node:url';

const root = fileURLToPath(new URL('..', import.meta.url));
const port = Number(process.env.PORT) || 5480;
const hosts = [`127.0.0.1:${port}`, `localhost:${port}`];
const types = {
  '.html': 'text/html; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8',
  '.mjs': 'text/javascript; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
  '.json': 'application/json; charset=utf-8',
  '.svg': 'image/svg+xml',
};

const server = createServer(async (req, res) => {
  // Only answer to our own address, so DNS rebinding can't reach the files.
  if (!hosts.includes(String(req.headers.host).toLowerCase())) {
    res.writeHead(403).end();
    return;
  }
  let path, file;
  try {
    path = decodeURIComponent(new URL(req.url, 'http://localhost').pathname);
    file = normalize(join(root, path === '/' ? 'index.html' : path));
  } catch {
    res.writeHead(400).end('Bad request');
    return;
  }
  // No path outside the project, and nothing dotted (.git, .gitignore, ..),
  // including Windows short names for them (GIT~1).
  if (!file.startsWith(root) || path.split(/[\\/]/).some((part) => part.startsWith('.') || part.includes('~'))) {
    res.writeHead(403).end();
    return;
  }
  try {
    const body = await readFile(file);
    res.writeHead(200, {
      'Content-Type': types[extname(file)] || 'application/octet-stream',
      'Cache-Control': 'no-store',
    });
    res.end(body);
  } catch {
    res.writeHead(404).end('Not found');
  }
});

server.on('error', (err) => {
  console.error(
    err.code === 'EADDRINUSE' ? `Port ${port} is already in use. Set PORT to use another.` : err.message,
  );
  process.exit(1);
});

server.listen(port, '127.0.0.1', () => console.log(`MathTime on http://127.0.0.1:${port}`));
```

```js
// The rules of play, with no DOM. The view asks it what happened and animates.
//
// `state` is one object, updated in place and never replaced: the view and the
// browser tests hold references to it.
//
// phase: 'playing' – accepting input
//        'correct' – answer was right; input locked until next()
//        'out'     – third miss; input locked until restart()
export function createGame(level, rng = Math.random) {
  const state = {
    phase: 'playing',
    question: 1,
    lives: level.livesPerProblem,
    typed: '',
    problem: level.makeProblem(rng),
  };

  function fresh() {
    state.problem = level.makeProblem(rng);
    state.lives = level.livesPerProblem;
    state.typed = '';
    state.phase = 'playing';
  }

  // One key from the keypad or keyboard: a digit, 'Backspace' or 'Enter'.
  // Returns 'typed' | 'ignored' | 'correct' | 'wrong' | 'out'.
  // 'typed' means the answer changed: a digit was added or removed.
  function press(key) {
    if (state.phase !== 'playing') return 'ignored';
    if (/^\d$/.test(key)) {
      if (state.typed.length >= level.maxAnswerLength) return 'ignored';
      state.typed += key;
      return 'typed';
    }
    if (key === 'Backspace') {
      if (!state.typed) return 'ignored';
      state.typed = state.typed.slice(0, -1);
      return 'typed';
    }
    if (key === 'Enter') {
      if (!state.typed) return 'ignored';
      if (Number(state.typed) === state.problem.answer) {
        state.phase = 'correct';
        return 'correct';
      }
      state.typed = '';
      state.lives -= 1;
      if (state.lives === 0) {
        state.phase = 'out';
        return 'out';
      }
      return 'wrong';
    }
    return 'ignored';
  }

  // Only a correct answer moves on; next() in any other phase does nothing.
  function next() {
    if (state.phase !== 'correct') return;
    state.question += 1;
    fresh();
  }

  function restart() {
    state.question = 1;
    fresh();
  }

  return { state, press, next, restart };
}
```

```js
// RGBA pixel data where paint(set) sets alpha values.
function pixels(width, height, paint) {
  const data = new Uint8ClampedArray(width * height * 4);
  paint((x, y, a) => { data[(y * width + x) * 4 + 3] = a; });
  return data;
}
```

```js
// Pure geometry for ink-based alignment (spec §6.1). No DOM, so it is
// unit-tested. Metrics are in em (fractions of the font size).

// Ink edges in RGBA pixel data (from getImageData), in canvas px. Each edge
// uses how much of its pixel is covered, for sub-pixel precision.
export function edgesFromAlpha(data, width, height, min = 4) {
  const rows = new Uint8Array(height);
  const cols = new Uint8Array(width);
  for (let y = 0; y < height; y++) {
    for (let x = 0; x < width; x++) {
      const a = data[(y * width + x) * 4 + 3];
      if (a > rows[y]) rows[y] = a;
      if (a > cols[x]) cols[x] = a;
    }
  }
  let t = 0, b = height - 1, l = 0, r = width - 1;
  while (t < height && rows[t] < min) t++;
  if (t === height) return null;
  while (rows[b] < min) b--;
  while (cols[l] < min) l++;
  while (cols[r] < min) r--;
  return {
    top: t + 1 - rows[t] / 255,
    bottom: b + rows[b] / 255,
    left: l + 1 - cols[l] / 255,
    right: r + cols[r] / 255,
  };
}

// Edges around a text origin (x0, y0) drawn at `size` px, as em metrics:
// ascent and descent from the baseline, ink left of / right of the origin
// (abl, abr), and the advance width.
export function toMetrics(edges, x0, y0, size, advance) {
  return {
    asc: (y0 - edges.top) / size,
    desc: (edges.bottom - y0) / size,
    abl: (x0 - edges.left) / size,
    abr: (edges.right - x0) / size,
    adv: advance / size,
  };
}

// Where text must start so its ink is centred across `width` em.
// off: the text indent (may be negative); right: where the ink ends; w: ink width.
export function centreInk(m, width) {
  const off = width / 2 - (m.abr - m.abl) / 2;
  return { off, right: off + m.abr, w: m.abl + m.abr };
}

// The ink's vertical span inside a 1em line box whose baseline is at `base` em.
export function band(m, base) {
  const top = base - m.asc;
  const bottom = base + m.desc;
  return { top, h: bottom - top, mid: (top + bottom) / 2 };
}

// Grow a span (CSS px) outward to whole device pixels, so it always covers the ink.
export function snapOutward(top, bottom, dpr) {
  return {
    top: Math.floor(top * dpr + 1e-3) / dpr,
    bottom: Math.ceil(bottom * dpr - 1e-3) / dpr,
  };
}

// Nearest whole device pixel, in CSS px.
export const snap = (v, dpr) => Math.round(v * dpr) / dpr;
```

```js
// screenshot pixels. Coverage comes from the blue channel: background #12407e

  const x0 = Math.max(0, Math.floor(area.left * dpr));
  const x1 = Math.min(png.width, Math.ceil(area.right * dpr));
  const y0 = Math.max(0, Math.floor(area.top * dpr));
  const y1 = Math.min(png.height, Math.ceil(area.bottom * dpr));
  const rows = new Float64Array(y1 - y0);
  const cols = new Float64Array(x1 - x0);
  for (let y = y0; y < y1; y++) {
    for (let x = x0; x < x1; x++) {
      const blue = png.data[(y * png.width + x) * 4 + 2];
      const f = Math.min(1, Math.max(0, (blue - 126) / (255 - 126)));
      if (f > rows[y - y0]) rows[y - y0] = f;
      if (f > cols[x - x0]) cols[x - x0] = f;
    }
  }
  const MIN = 0.05;
  let t = 0, b = rows.length - 1, l = 0, r = cols.length - 1;
  while (t < rows.length && rows[t] < MIN) t++;
  if (t === rows.length) return null;

  while (cols[l] < MIN) l++;
  while (cols[r] < MIN) r--;
```

```js
import { edgesFromAlpha, toMetrics } from './ink.js';

// Measures a string's ink by drawing it offscreen at its real on-screen size
// in device pixels, so font hinting matches what is drawn (spec §6.1).
// style: { weight, family } as in computed CSS. Returns em metrics (ink.js).
export function createScanner() {
  const canvas = document.createElement('canvas');
  const ctx = canvas.getContext('2d', { willReadFrequently: true });
  const cache = new Map();
  return function scan(text, style, size) {
    const key = `${style.weight}|${style.family}|${size}|${text}`;
    if (cache.has(key)) return cache.get(key);
    const width = Math.ceil(size * (text.length * 1.3 + 1));
    const height = Math.ceil(2 * size);
    const x0 = Math.round(size / 2);
    const y0 = Math.round(size * 1.3);
    canvas.width = width;              // also clears the canvas
    canvas.height = height;
    ctx.font = `${style.weight} ${size}px ${style.family}`;
    ctx.textBaseline = 'alphabetic';
    ctx.fillText(text, x0, y0);
    const edges = edgesFromAlpha(ctx.getImageData(0, 0, width, height).data, width, height);
    const m = toMetrics(edges, x0, y0, size, ctx.measureText(text).width);
    cache.set(key, m);
    return m;
  };
}
```

```js
import { centreInk, band, snapOutward, snap } from './ink.js';

// Measures the real ink of the equation's font and lays the screen out around
// it (spec §6.1): box, caret and operators from measured ink; equation fitted
// between the gutters and snapped to device pixels; header fitted; keypad
// labels centred. `afterMeasure` runs after every measurement so the view can
// re-render digits with the new geometry.
export function createLayout(dom, scan, possibleAnswers) {
  const { eq, slot, caret, stage, topBar } = dom;
  const geo = { fs: 0, style: null, base: 0.8, mid: 0.5, boxW: 1.2, wide: 0.6 };
  const api = { geo, inkOf, layout, afterMeasure: null };

  // Ink geometry of `text` centred across `width` em, at the current size.
  function inkOf(text, width) {
    return centreInk(scan(text, geo.style, geo.fs * devicePixelRatio), width);
  }

  // Baseline position (em) inside a 1em, line-height 1 box, read from layout.
  function domBaseline() {
    const box = document.createElement('span');
    box.style.cssText = 'position:absolute;visibility:hidden;display:block;height:1em;line-height:1;white-space:pre';
    box.innerHTML = '0<span style="display:inline-block;width:0;height:0;vertical-align:baseline"></span>';
    eq.appendChild(box);
    const base = (box.lastChild.getBoundingClientRect().top - box.getBoundingClientRect().top) / geo.fs;
    box.remove();
    return base;
  }

  function measure() {
    const cs = getComputedStyle(eq);
    geo.fs = parseFloat(cs.fontSize);
    geo.style = { weight: cs.fontWeight, family: cs.fontFamily };
    const size = geo.fs * devicePixelRatio;
    geo.base = domBaseline();
    const digits = band(scan('0123456789', geo.style, size), geo.base);
    geo.mid = digits.mid;
    // Fonts' "tabular" digits can still differ in width; use the widest.
    geo.wide = Math.max(...[...'0123456789'].map(c => scan(c, geo.style, size).adv));
    geo.boxW = Math.max(...possibleAnswers.map(s => inkOf(s, 0).w));
    eq.style.setProperty('--fs', geo.fs + 'px');
    eq.style.setProperty('--ink-top', digits.top.toFixed(4));
    eq.style.setProperty('--ink-h', digits.h.toFixed(4));
    eq.style.setProperty('--wide', geo.wide.toFixed(4));
    eq.style.setProperty('--box-w', geo.boxW.toFixed(4));
    eq.style.setProperty('--bw', getComputedStyle(slot, '::after').borderTopWidth);
    for (const op of eq.querySelectorAll('.op')) {
      const opMid = band(scan(op.textContent, geo.style, size), geo.base).mid;
      op.style.top = (digits.mid - opMid).toFixed(4) + 'em';
    }
    // Centre the digits' ink, not the line box, in the problem area.
    eq.style.marginTop = ((0.5 - digits.mid) * 2).toFixed(4) + 'em';
    // The caret spans the digit ink, snapped outward to whole device pixels
    // around the (pixel-aligned) baseline, so it never comes up short.
    const cap = snapOutward((digits.top - geo.base) * geo.fs, (digits.top + digits.h - geo.base) * geo.fs, devicePixelRatio);
    caret.style.top = (geo.base * geo.fs + cap.top).toFixed(3) + 'px';
    caret.style.height = (cap.bottom - cap.top).toFixed(3) + 'px';
    api.afterMeasure?.();
  }

  // The box is last in the equation, so its dashed construction outline
  // sticks out on the right. Count it when fitting and centring.
  function overhang() {
    const after = getComputedStyle(slot, '::after');
    return after.outlineStyle === 'none' ? 0 : parseFloat(after.outlineOffset) + parseFloat(after.outlineWidth);
  }

  // Drop breadcrumb parts from the left until the header fits on one line.
  function fitHeader() {
    topBar.classList.remove('tight', 'tighter');
    const where = topBar.querySelector('.where');
    const crumbs = topBar.querySelector('.crumbs');
    if (crumbs.scrollWidth > where.clientWidth) topBar.classList.add('tight');
    if (crumbs.scrollWidth > where.clientWidth) topBar.classList.add('tighter');
  }

  // Centre each digit key's label on its real ink. Labels are much smaller
  // than the equation and snap to pixels differently, so each is scanned at
  // its own device size. Text baselines snap to whole device pixels, so the
  // vertical shift is a whole number of device pixels too.
  function centreKeys() {
    const dpr = devicePixelRatio;
    for (const kl of document.querySelectorAll('.pad .kl')) {
      const kr = kl.parentElement.getBoundingClientRect();
      if (!kr.width) continue;
      kl.style.transform = 'none';
      const cs = getComputedStyle(kl);
      const kfs = parseFloat(cs.fontSize);
      const m = scan(kl.textContent, { weight: cs.fontWeight, family: cs.fontFamily }, kfs * dpr);
      const probe = document.createElement('span');
      probe.style.cssText = 'display:inline-block;width:0;height:0;vertical-align:baseline';
      kl.appendChild(probe);
      const baseY = snap(probe.getBoundingClientRect().top, dpr);
      probe.remove();
      const lr = kl.getBoundingClientRect();
      const dy = snap(kr.top + kr.height / 2 - (baseY - (m.asc - m.desc) / 2 * kfs), dpr);
      const dx = kr.left + kr.width / 2 - (m.abr - m.abl) / 2 * kfs - lr.left;
      kl.style.transform = `translate(${dx.toFixed(3)}px, ${dy.toFixed(3)}px)`;
    }
  }

  function layout() {
    if (!stage.clientWidth) return;            // hidden (level selector showing)
    fitHeader();
    eq.style.fontSize = '';
    eq.style.left = '';
    eq.style.top = '';
    measure();
    const gutter = innerWidth <= 480 ? 16 : 24;
    const avail = stage.clientWidth - 2 * gutter - 1;   // 1px slack for pixel snapping
    for (let i = 0; i < 3; i++) {                       // the overhang scales with the font
      const width = eq.getBoundingClientRect().width + overhang();
      if (width <= avail) break;
      eq.style.fontSize = (geo.fs * avail / width) + 'px';
      measure();
    }
    // Baseline and left edge on whole device pixels; shift left by half the
    // overhang so the visible group (equation plus outline) is centred.
    const dpr = devicePixelRatio;
    const r = eq.getBoundingClientRect();
    eq.style.setProperty('--overhang', overhang() + 'px');
    eq.style.left = (snap(r.left - overhang() / 2, dpr) - r.left).toFixed(3) + 'px';
    const baseY = r.top + geo.base * geo.fs;
    eq.style.top = (snap(baseY, dpr) - baseY).toFixed(3) + 'px';
    centreKeys();
  }

  return api;
}
```

```js
export function createView(dom, layout, level) {
  const { slot, shown, caret, qnum, hearts, livesBox, a, b } = dom;

  function renderDigits(el, str) {

  // Answer: ink-centred in the fixed box, caret a fixed gap after the ink.
  // A wrong entry wider than the box is scaled down around its centre.
  function renderAnswer(typed) {
    const { geo } = layout;
    const span = document.createElement('span');
    span.className = 'cell';
    span.style.width = 'calc(var(--box-w) * 1em)';
    span.textContent = typed;
    const g = typed ? layout.inkOf(typed, geo.boxW) : { off: 0, right: 0, w: 0 };
    span.style.textIndent = g.off.toFixed(4) + 'em';
    if (g.w > geo.boxW + 1e-3) {
      span.style.transform = `scale(${(geo.boxW / g.w).toFixed(4)})`;
      span.style.transformOrigin = `50% ${geo.mid.toFixed(4)}em`;
    }
    shown.replaceChildren(span);
    caret.style.left = typed ? `calc(${g.right.toFixed(4)}em + max(0.08em, 4px))` : '';
    slot.classList.toggle('empty', typed === '');
    slot.classList.toggle('full', typed.length >= level.maxAnswerLength);
  }

  function render(state) {
    renderDigits(a, String(state.problem.a));
    renderDigits(b, String(state.problem.b));
    renderAnswer(state.typed);
    qnum.textContent = 'Question ' + state.question;
    hearts.forEach((h, i) => h.classList.toggle('lost', i >= state.lives));
    livesBox.setAttribute('aria-label', state.lives === 1 ? '1 life left' : `${state.lives} lives left`);
  }

  return { render, renderAnswer };
}
```

```js
export function bindInput({ ans, pad, slot }, onKey) {
  ans.addEventListener('keydown', e => {
    e.preventDefault();
    if (e.key !== 'Enter' && e.key !== 'Backspace' && !/^\d$/.test(e.key)) return;
    if (onKey(e.key) === 'ignored') return;
    const btn = pad.querySelector(`button[data-k="${e.key}"]`);
    btn.classList.add('pressed');
    setTimeout(() => btn.classList.remove('pressed'), 110);
  });
  // All typing goes through keydown; keep the hidden input empty.
  ans.addEventListener('input', () => { ans.value = ''; });
  ans.addEventListener('focus', () => slot.classList.add('focused'));
  ans.addEventListener('blur', () => slot.classList.remove('focused'));

  pad.addEventListener('click', e => {
    const btn = e.target.closest('button[data-k]');
    if (btn) onKey(btn.dataset.k);
  });

  // Keep focus in the answer while playing, without letting taps steal it.
  document.addEventListener('pointerdown', e => {
    if (!document.body.classList.contains('at-select') && e.target !== ans) {
      e.preventDefault();
      ans.focus({ preventScroll: true });
```

```js
  // A wrong entry wider than the box is scaled down around its centre.
  function renderAnswer(typed) {
    const { geo } = layout;
    const span = document.createElement('span');
    span.className = 'cell';
    span.style.width = 'calc(var(--box-w) * 1em)';
    span.textContent = typed;
    const g = typed ? layout.inkOf(typed, geo.boxW) : { off: 0, right: 0, w: 0 };
    span.style.textIndent = g.off.toFixed(4) + 'em';
    if (g.w > geo.boxW + 1e-3) {
      span.style.transform = `scale(${(geo.boxW / g.w).toFixed(4)})`;
      span.style.transformOrigin = `50% ${geo.mid.toFixed(4)}em`;
    }
    shown.replaceChildren(span);
    caret.style.left = typed ? `calc(${g.right.toFixed(4)}em + max(0.08em, 4px))` : '';
    slot.classList.toggle('empty', typed === '');
    slot.classList.toggle('full', typed.length >= level.maxAnswerLength);
  }

  function render(state) {
    renderDigits(a, String(state.problem.a));
    renderDigits(b, String(state.problem.b));
    renderAnswer(state.typed);
    qnum.textContent = 'Question ' + state.question;
    hearts.forEach((h, i) => h.classList.toggle('lost', i >= state.lives));
    livesBox.setAttribute('aria-label', state.lives === 1 ? '1 life left' : `${state.lives} lives left`);
  }

  function play(cls) {
    eq.classList.remove(...ANIMS);
    void eq.offsetWidth;                 // restart the animation
    eq.classList.add(cls);
  }
  eq.addEventListener('animationend', e => {
    // 'out' holds its end state until 'in' replaces it.
    if (e.target === eq && e.animationName !== 'out') eq.classList.remove(...ANIMS);
  });

  // Green for 0.35s, slide out, swap in the next problem, slide in.
  async function showCorrect(state, next) {
    eq.classList.add('correct');
    await wait(350);
    play('out');
    await wait(220);
    next();
    eq.classList.remove('correct');
    render(state);
    play('in');
  }

  function showWrong(state) {
    render(state);
    play('shake');
  }

  // Third miss: shake, then the level selector 0.45s later.
  async function showOut(state) {
    render(state);
    play('shake');
    await wait(450);
    eq.classList.remove(...ANIMS);
    document.body.classList.add('at-select');
  }

  return { render, renderAnswer, showCorrect, showWrong, showOut };
}
```
