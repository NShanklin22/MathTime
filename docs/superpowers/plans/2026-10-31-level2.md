# MathTime Level 2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Level 2 (subtraction within 10), the level selector the game now opens on, and the breadcrumb's way back, exactly as `docs/superpowers/specs/2026-10-21-level2-design.md` describes (approved by the user 2026-10-31).

**Architecture:** The game stays one static page of plain ES modules with no build step.
- Each level is a plain object (`level1.js`, `level2.js`), listed in route order in `levels.js`.
- The game switches level with `game.start(level)`.
- One `play(level)` in `main.js` tells the layout and view about the level and lays the screen out again: the operator and the answer box belong to the level.
- The selector is a new module, `selector.js`. It builds a row per level, marks the level you came from, and lines its text and samples up on their measured ink, as `layout.js` does for the equation.
- `body.at-select` switches between the selector and the play screen, as it already does for Level 1's bare button.

**Tech Stack:** HTML, CSS, JavaScript (ES modules). `node:test` for unit tests. `@playwright/test` 1.55.1 and `pngjs` for browser tests (the phone project is a real 2× window, the desktop project 1440×900 at 1×). IBM Plex Mono from Google Fonts.

Both passed UI review. Serve them with `npm start` and open them at `http://127.0.0.1:5480/docs/superpowers/specs/…`. The code below ports what they do into the game.

```js
// The rules of play, with no DOM. The view asks it what happened and animates.
//
// `state` is one object, updated in place and never replaced: the view and the
// browser tests hold references to it. `level` is the level being played;
// start() switches it.
//
// phase: 'playing' – accepting input
//        'correct' – answer was right; input locked until next() or start()
//        'out'     – third miss; input locked until restart() or start()
export function createGame(level, rng = Math.random) {
  let current = level;
  const state = {
    phase: 'playing',
    question: 1,
    lives: current.livesPerProblem,
    typed: '',
    problem: current.makeProblem(rng),
  };

  function fresh() {
    state.problem = current.makeProblem(rng);
    state.lives = current.livesPerProblem;
    state.typed = '';
    state.phase = 'playing';
  }

  // One key from the keypad or keyboard: a digit, 'Backspace' or 'Enter'.
  // Returns 'typed' | 'ignored' | 'correct' | 'wrong' | 'out'.
  // 'typed' means the answer changed: a digit was added or removed.
  function press(key) {
    if (state.phase !== 'playing') return 'ignored';
    if (/^\d$/.test(key)) {
      if (state.typed.length >= current.maxAnswerLength) return 'ignored';
      state.typed += key;
      return 'typed';
    }
    if (key === 'Backspace') {
      if (!state.typed) return 'ignored';
      state.typed = state.typed.slice(0, -1);
      return 'typed';
    }
    if (key === 'Enter') {
      if (!state.typed) return 'ignored';
      if (Number(state.typed) === state.problem.answer) {
        state.phase = 'correct';
        return 'correct';
      }
      state.typed = '';
      state.lives -= 1;
      if (state.lives === 0) {
        state.phase = 'out';
        return 'out';
      }
      return 'wrong';
    }
    return 'ignored';
  }

  // Only a correct answer moves on; next() in any other phase does nothing.
  function next() {
    if (state.phase !== 'correct') return;
    state.question += 1;
    fresh();
  }

  // Begin `chosen` (the level being played, by default) at question 1, with
  // full lives and a new problem, whatever the phase.
  function start(chosen = current) {
    current = chosen;
    state.question = 1;
    fresh();
  }

  function restart() {
    start();
  }

  return { state, press, next, start, restart, get level() { return current; } };
}
```

```js
export async function boxInside(page) {
```

```js
// The level selector (Level 2 spec §3): a row per built level, grouped by
// topic in route order, each a button that calls choose(level). The level
// you came from is marked with an amber bar and gets the focus.
export function createSelector(root, levels, choose) {
  const el = (tag, cls, text) => {
    const node = document.createElement(tag);
    if (cls) node.className = cls;
    if (text != null) node.textContent = text;
    return node;
  };
  const rows = new Map();                    // level → its row
  let group = null;
  for (const level of levels) {
    if (group?.dataset.topic !== level.topic) {
      group = el('section', 'group');
      group.dataset.topic = level.topic;
      group.append(el('h2', 'group-name', level.topic));
      root.append(group);
    }
    const row = el('button', 'row');
    row.dataset.level = level.n;
    row.setAttribute('aria-label', `Level ${level.n}: ${level.name}`);
    const txt = el('span', 'txt');
    txt.append(el('span', 'title', `Level ${level.n}`), el('span', 'name', level.name));
    // A sample problem, set like the equation: digits, then the operator.
    const sample = el('span', 'sample');
    sample.setAttribute('aria-hidden', 'true');
    const [a, b] = level.sample;
    sample.append(el('span', null, String(a)), el('span', 'op', level.op), el('span', null, String(b)));
    row.append(txt, sample);
    row.addEventListener('click', () => choose(level));
    rows.set(level, row);
    group.append(row);
  }

  // The Enter that made a third miss may still be held: its key repeat
  // (about 0.5s) would otherwise press a row as soon as it has focus.
  root.addEventListener('keydown', e => {
    if (e.repeat && (e.key === 'Enter' || e.key === ' ')) e.preventDefault();
  });

  // Mark the level you came from, or none.
  function mark(level) {
    for (const [l, row] of rows) row.classList.toggle('current', l === level);
  }

  // Focus the marked row, so the keyboard carries on from there.
  function focus() {
    root.querySelector('.row.current')?.focus({ preventScroll: true });
  }

  return { mark, focus };
}
```

```js
import { band, snap, onLayoutGrid } from './ink.js';

const DIGITS = [...'0123456789'];

// The level selector (Level 2 spec §3): a row per built level, grouped by
// topic in route order, each a button that calls choose(level). The level
// you came from is marked with an amber bar and gets the focus. layout()
// lines the labels and samples up on their measured ink (spec §5), as
// layout.js does the equation; it can run any number of times.
export function createSelector(root, levels, scan, choose) {
  const el = (tag, cls, text) => {
    const node = document.createElement(tag);
    if (cls) node.className = cls;
    if (text != null) node.textContent = text;
    return node;
  };
  const rows = new Map();                    // level → its row
  let group = null;
  for (const level of levels) {
    if (group?.dataset.topic !== level.topic) {
      group = el('section', 'group');
      group.dataset.topic = level.topic;
      group.append(el('h2', 'group-name', level.topic));
      root.append(group);
    }
    const row = el('button', 'row');
    row.dataset.level = level.n;
    row.setAttribute('aria-label', `Level ${level.n}: ${level.name}`);
    const txt = el('span', 'txt');
    txt.append(el('span', 'title', `Level ${level.n}`), el('span', 'name', level.name));
    // A sample problem, set like the equation: digits, then the operator.
    const sample = el('span', 'sample');
    sample.setAttribute('aria-hidden', 'true');
    const [a, b] = level.sample;
    sample.append(el('span', null, String(a)), el('span', 'op', level.op), el('span', null, String(b)));
    row.append(txt, sample);
    row.addEventListener('click', () => choose(level));
    rows.set(level, row);
    group.append(row);
  }

  // The Enter that made a third miss may still be held: its key repeat
  // (about 0.5s) would otherwise press a row as soon as it has focus.
  root.addEventListener('keydown', e => {
    if (e.repeat && (e.key === 'Enter' || e.key === ' ')) e.preventDefault();
  });

  // Mark the level you came from, or none.
  function mark(level) {
    for (const [l, row] of rows) row.classList.toggle('current', l === level);
  }

  // Focus the marked row, so the keyboard carries on from there.
  function focus() {
    root.querySelector('.row.current')?.focus({ preventScroll: true });
  }

  // A text element's font as the scanner wants it, and its text as drawn.
  function textOf(node) {
    const cs = getComputedStyle(node);
    const fs = parseFloat(cs.fontSize);
    const style = { weight: cs.fontWeight, family: cs.fontFamily, color: cs.color, spacing: (parseFloat(cs.letterSpacing) || 0) / fs };
    return { fs, style, text: cs.textTransform === 'uppercase' ? node.textContent.toUpperCase() : node.textContent };
  }

  // Move a label so its ink, not its box, starts at its left edge (a
  // heading's at the column's, a row's text at its padding), the start on a
  // whole device pixel, as the breadcrumb's (header.js). It is placed from
  // where its box starts unmoved, so placing it again changes nothing.
  function inkLeft(node) {
    const dpr = devicePixelRatio;
    node.style.marginLeft = '';
    const { fs, style, text } = textOf(node);
    const left = node.getBoundingClientRect().left;
    const at = snap(left + scan(text, style, fs * dpr).abl * fs, dpr);
    node.style.marginLeft = onLayoutGrid(at - left, dpr) + 'px';
  }

  // A sample, set as the equation is (layout.js): the baseline from a probe,
  // the digit band from all ten digits' ink, each operator moved onto the
  // band's centre, and the whole sample moved so that centre is its box's,
  // which the row centres. Then across, so its last digit's ink ends where
  // its box does, on the row's right padding edge. Its own rects include the
  // move already made, which cancels out, so placing it again changes nothing.
  function placeSample(sample) {
    const { fs, style } = textOf(sample);
    const size = fs * devicePixelRatio;
    const probe = document.createElement('span');
    probe.style.cssText = 'position:absolute;visibility:hidden;display:block;height:1em;line-height:1;white-space:pre';
    probe.innerHTML = '0<span style="display:inline-block;width:0;height:0;vertical-align:baseline"></span>';
    sample.appendChild(probe);
    const base = (probe.lastChild.getBoundingClientRect().top - probe.getBoundingClientRect().top) / fs;
    probe.remove();
    const each = DIGITS.map(c => scan(c, style, size));
    const digits = band({ asc: Math.max(...each.map(m => m.asc)), desc: Math.max(...each.map(m => m.desc)) }, base);
    for (const op of sample.querySelectorAll('.op')) {
      const m = scan(op.textContent, { ...style, color: getComputedStyle(op).color }, size);
      op.style.top = (digits.mid - band(m, base).mid).toFixed(4) + 'em';
    }
    sample.style.setProperty('--ink-dy', (0.5 - digits.mid).toFixed(4) + 'em');
    const last = sample.lastElementChild;
    const inkRight = last.getBoundingClientRect().left + scan(last.textContent, style, size).abr * fs;
```
